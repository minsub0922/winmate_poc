class Component extends DCLogic {
 renderVals() {
 const steps = ['넣기', '경쟁사 확인', '분석', '결과'];
 // 입력 "카페 메뉴보드 사이니지" 에서 읽은 것: 업종 찾음 · 제품 일부 · 고객사 · 장소 비어 있음(점선)
 const chip = (found) => '' + (found ? '' : '');
 const C = [['업종', '외식 · 카페', true, false], ['제품', '메뉴보드 사이니지', true, true], ['고객사', '', false, false], ['장소', '', false, false]];
 const chips = C.map(([k, v, found, partial]) => ({ k, v: found ? v : '비어 있음', found, partial, style: chip(found) }));
 const n = chips.filter((c) => c.found).length;
 return { steps, cur: 1, chips, foundCount: n + ' / ' + chips.length };
 }
}
