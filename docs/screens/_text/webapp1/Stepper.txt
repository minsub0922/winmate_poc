class Component extends DCLogic {
 renderVals() {
 const steps = Array.isArray(this.props.steps) ? this.props.steps : ['1단계', '2단계', '3단계'];
 const cur = Number(this.props.current ?? 2);
 const complete = !!this.props.complete;
 const autoFrom = Number(this.props.autoFrom ?? 0);
 const n = steps.length;
 // 딸깍 버튼: 켜져 있고 아직 남은 단계가 있을 때만
 const oneClick = !!this.props.oneClick;
 const showBtn = oneClick && !complete && cur 0);
 const open = showBtn && (!!this.props.oneClickOpen || !!(this.state && this.state.open));
 const gap = n > 5 ? 8 : 12;
 const lineW = n > 5 ? 28 : (n > 3 ? 40 : 56);
 const circleBase = '';
 const items = steps.map((text, i) => {
 const k = i + 1;
 const done = complete || k 0 && k >= autoFrom;
 return {
 text, n: k, auto, check: done && !auto, num: !done, notLast: i 5 ? '' : '') + '; white-space: nowrap;' + (active ? '' : done ? '' : ''),
 line: 'width: ' + lineW + '' + gap + '' + (complete || k 0;
 const rows = given ? this.props.planRows : steps.slice(cur - 1).map((label, i) => ({ label, note: i === 0 ? '입력한 내용까지 반영하고 나머지 추론' : '추론해 자동 완성' }));
 const confirmed = Array.isArray(this.props.planConfirmed) ? this.props.planConfirmed : steps.slice(0, cur - 1);
 return {
 items, showBtn, open, rows, confirmed, hasConfirmed: confirmed.length > 0, yes: true,
 summary: this.props.planSummary || ('남은 ' + rows.length + '단계'),
 rootStyle: ''Noto Sans KR\'' + (showBtn ? '' : ' justify-content: center;'),
 btnStyle: '' + (open ? '' : ''),
 toggle: () => this.setState({ open: !open }),
 close: () => this.setState({ open: false })
 };
 }
}
