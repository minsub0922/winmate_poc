class Component extends DCLogic {
 renderVals() {
 // 작업: 제목 · 담당/시점 · 입력 방식(free 자유 양식 · req 요구사항에서) · 경쟁사 수 · 상태(done 완료 · run 확인 중 · upd 업데이트 필요) · 상태 메모 · 보낸 곳 · 열기 라벨 · 열기 화면
 const R = [
 ['A 커피 메뉴보드 경쟁사 분석', '최민섭 · 2시간 전 수정', 'free', '4', '곳', 'done', '출처 16 · 삼성 강점 3', 'MI 작업 · 제안서', '결과 보기', 'CA4.dc.html'],
 ['B 병원 안내 사이니지 경쟁사', '최민섭 · 오늘 시작', 'req', '5', '후보', 'run', '추천 3 · 확인 필요 2', '', '이어서', 'CA2.dc.html'],
 ['C 물류센터 관제 디스플레이 경쟁사', '최민섭 · 9월 2일 분석', 'free', '3', '곳', 'upd', '완료 · 경쟁사 B 신제품 2026.09', 'Storyboard', '상세 보기', 'CA4D.dc.html']
 ];
 const ST = { done: '완료', run: '확인 중', upd: '업데이트 필요' };
 const CHIP = {
 done: '',
 run: '',
 upd: ''
 };
 const IN = { free: '자유 양식', req: '요구사항에서' };
 const cols = '';
 const rows = R.map(([title, sub, inp, count, countUnit, s, note, sent, act, open]) => ({
 title, sub, count, countUnit, note, sent, act, open,
 input: IN[inp], isFree: inp === 'free', isReq: inp === 'req',
 hasSent: sent.length > 0, noSent: sent.length === 0,
 stText: ST[s], isDone: s === 'done', isRun: s === 'run', isUpd: s === 'upd',
 rowStyle: 'display: grid; ' + cols + '',
 inputStyle: '',
 countStyle: '' + (s === 'run' ? '' : ''),
 chipStyle: '' + CHIP[s],
 actStyle: ''
 }));
 const count = (s) => R.filter((r) => r[5] === s).length;
 const F = [['전체', R.length, true], ['완료', count('done') + count('upd'), false], ['확인 중', count('run'), false]];
 const filters = F.map(([label, n, on]) => ({
 label, n, on,
 style: '' + (on ? '' : '')
 }));
 const starts = [['고객 요구사항 정의서에서', '3건', 'CA1R.dc.html'], ['MI 작업의 경쟁사에서', '3건', 'MI2C.dc.html']].map(([name, n, href]) => ({ name, n, href }));
 return {
 rows, filters, starts, total: R.length, updCount: count('upd'),
 headStyle: 'display: grid; ' + cols + ''
 };
 }
}
