class Component extends DCLogic {
 renderVals() {
 const steps = ['넣기', '경쟁사 확인', '분석', '결과'];
 // 작업 줄 상태: done(완료 · 채운 원) / active(진행 중 · 점 테두리) / wait(대기 · 연한 테두리)
 const dot = (s) => '' + (s === 'done' ? '' : s === 'active' ? '' : '');
 const text = (s) => '' + (s === 'active' ? '' : s === 'done' ? '' : '');
 const tag = (s) => '' + (s === 'active' ? '' : '');
 const LABEL = { done: '완료', active: '진행 중', wait: '대기' };
 const jobs = [['입력 읽기 — 고객사 · 업종 · 장소 · 제품', 'done'], ['같은 업종 도입사례 23건에서 자주 비교된 곳', 'done'], ['수도권 · 55" 사이니지 공개 자료 검색', 'active'], ['후보 정리 · 근거 붙이기', 'wait']]
 .map(([t, s]) => ({ t, state: LABEL[s], done: s === 'done', active: s === 'active', dot: dot(s), text: text(s), tag: tag(s) }));
 return { steps, cur: 1, jobs };
 }
}
