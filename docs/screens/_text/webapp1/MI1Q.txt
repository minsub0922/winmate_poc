class Component extends DCLogic {
 renderVals() {
 // 회의록에서 읽힌 단서와 그 단서가 가리키는 업종
 const C = [['로비', 'HT'], ['투숙객', 'HT'], ['객실 320실', 'HT'], ['라운지 카페', 'FB'], ['메뉴보드', 'FB'], ['음료 피크타임', 'FB']];
 const clues = C.map(([t, code]) => ({
 t, code,
 tagStyle: '' + (code === 'HT' ? '' : '')
 }));
 // 선택지: [이름, 설명, 점수, 레이아웃 코드, 썸네일 3, 추천]
 const O = [
 ['호텔 · 리조트', '호텔 운영 관점 — 로비 · 객실 경험, 투숙객 여정, 호텔 매출 구조. 사례 17건', 0.58, 'MI-HT-A · B · C', ['barline', 'process', 'journey'], true],
 ['외식 · 카페', '매장 운영 관점 — 메뉴 · 피크 주문, 손님 체류, 가맹 구조. 사례 23건', 0.55, 'MI-FB-A · B · C', ['barline', 'process', 'journey'], false],
 ['섞어서 보기', '시장 · 비즈니스는 호텔로, 사용자 여정은 라운지 카페 손님으로', 0.57, 'MI-HT-A · B + MI-FB-C', ['barline', 'process', 'journey'], false]
 ];
 const opts = O.map(([name, desc, s, codes, thumbs, rec]) => ({
 name, desc, codes, rec, on: rec,
 thumbs: thumbs.map((kind) => ({ kind, n: 3 })),
 score: s.toFixed(2),
 bar: 'width: ' + Math.round(s * 100) + '' + (rec ? '' : '') + ';',
 box: '' + (rec ? '' : '')
 }));
 const done = ['쓰임 · 표준 MI 3시트', '범위 · 4개 영역', '경쟁사 · 익명 A · B · C', '사내 자료 · 호텔 17 + 외식 23건'];
 return { steps: ['고객 요구사항', '분석 범위', '분석 결과'], cur: 2, clues, opts, done };
 }
}
