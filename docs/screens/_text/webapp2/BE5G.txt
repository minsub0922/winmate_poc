 renderVals() {
 // [단계, 내용, 상태: done · run · wait]
 const R = [
 ['공간 구조', '120평 · 층고 4.5m · 전면 유리창 · 기둥 2개', 'done'],

 ];

 const rows = R.map(([name, note, s]) => ({
 name, note, st: ST[s],
 isDone: s === 'done', isRun: s === 'run', isWait: s === 'wait',
 rowStyle: '' + (s === 'run' ? '' : ''),
 stStyle: '' + (s === 'run' ? '' : s === 'done' ? '' : '')
 }));
 return { steps: ['공간 입력', '배치될 제품', '가구 추천', '배치 · 인테리어 컨펌', '3D 조감도 생성'], cur: 5, rows };
 }
