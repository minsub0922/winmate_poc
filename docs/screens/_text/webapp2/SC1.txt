 새로 만들기 
 시나리오 공간으로 조감도 생성 
 기존 조감도 연결 
 조감도 작업 3개에서 고르기 
 공간 시나리오 입력 
class Component extends DCLogic {
 renderVals() {
 // 조감도 추가: 기본 꺼짐. 켜면 조감도를 새로 만들지, 기존 조감도 작업에 연결할지 고른다.
 const st = this.state || {};
 const on = st.aerial === true;
 const src = st.src === 'exist' ? 'exist' : 'new';
 const opt = (sel) => '' + (sel ? '' : '');
 return {
 steps: ['시나리오 유형', '공간 시나리오 입력', '솔루션 · 제품 입력', '시나리오 생성'], cur: 1,
 aerialOn: on,
 aerialPressed: on ? 'true' : 'false',
 toggleAerial: () => this.setState({ aerial: !on }),
 aerialTag: on ? '켜짐' : '선택 · 기본 꺼짐',
 aerialTagStyle: '' + (on ? '' : ''),
 iconBoxStyle: '' + (on ? '' : '') + ';',
 iconStroke: on ? '' : '',
 toggleStyle: '' + (on ? '' : '') + ';',
 knobStyle: '' + (on ? '' : '') + '',
 pickNew: () => this.setState({ src: 'new' }),
 pickExist: () => this.setState({ src: 'exist' }),
 newPressed: src === 'new' ? 'true' : 'false',
 existPressed: src === 'exist' ? 'true' : 'false',
 newStyle: opt(src === 'new'),
 existStyle: opt(src === 'exist')
 };
 }
}
