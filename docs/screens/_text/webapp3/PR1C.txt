class Component extends DCLogic {
 renderVals() {
 // 원본 출처 탭 · 파일 올리기 선택
 const tabs = [['내 제안서 목록', false], ['파일 올리기', true]].map(([label, on]) => ({
 label, on,
 style: '' + (on ? '' : '')
 }));
 // 올린 파일 처리 단계: [라벨, 상태(done · busy)]
 const phases = [['읽기', 'done'], ['쪽 나누기', 'done'], ['기준별 분석 · 진행 중', 'busy']].map(([label, st]) => ({
 label, done: st === 'done', busy: st === 'busy',
 style: '' + (st === 'done' ? '' : '')
 }));
 // 내 제안서 목록 추천 2개
 const recs = [['A 커피 프랜차이즈 2025 매장 사이니지 제안', '같은 고객'], ['C 편의점 2026 메뉴보드 제안', '같은 솔루션']].map(([name, why]) => ({ name, why }));
 // 활용 방식: [키, 라벨, 설명, 작은 줄, 추천 여부]
 const M = [
 ['a', '기반으로 개선 · 수정', '내용까지 가져와 현행화 · 보강 · 재작성. 같은 고객의 지난 제안서에 알맞아요', '시트별 유지 · 갱신 · 재작성 · 신규 · 제외를 정해요', false],
 ['b', '논리 흐름만 차용', '설득 구조(스토리라인 · 시트 역할 · 주장 순서)만 가져오고 내용은 이번 고객으로 새로 써요. 다른 고객 · 동료 제안서에 알맞아요', '고객 정보 · 사진 · 수치는 쓰지 않아요', false],
 ['auto', '분석 뒤 추천받기', '9가지 기준 분석 결과를 보고 에이전트가 추천해요 · 같은 고객 + 요구사항 겹침 3건 이상이면 개선 · 수정, 다른 고객이면 흐름 차용', '', true]
 ];
 const picked = 'auto';
 const modes = M.map(([k, label, desc, sub, rec]) => {
 const on = k === picked;
 return {
 label, desc, sub, rec, on, hasSub: !!sub,
 style: '' + (on ? '' : ''),
 dotStyle: '' + (on ? '' : '')
 };
 });
 return {
 steps: ['고객 · 프로젝트', '제안서 유형', '시트 구성', '섹션 작성', '디자인 템플릿', 'PPTX 생성'], cur: 1,
 tabs, phases, recs, modes
 };
 }
}
